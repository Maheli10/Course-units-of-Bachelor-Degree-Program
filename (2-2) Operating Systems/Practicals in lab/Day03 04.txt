8.code:- (print all elements in array)

9.code:- (sum of elemets,sorting elements)
	numbers=(10 5 40 7 80)
	sum=0
	for num in "${numbers[@]}";
	do
	((sum +=num))
	done
	echo "Sum of numbers is $sum"

	sorted=($(printf '%s\n' "${numbers[@]}" | sort -n))
	echo "Sorted array: "
	for num in "${sorted[@]}";
	do
	echo $num
	done
output:-
	Sum of numbers is 142
	Sorted array: 
	5
	7
	10 
	40
	80
