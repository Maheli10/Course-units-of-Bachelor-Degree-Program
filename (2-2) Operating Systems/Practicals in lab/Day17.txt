sum(){
	i=0
	while [ $i -lt 4 ];
	do
	n=$(($num%10))
	sum=$(($sum+$n))
	num=$(($num/10))
	((i++))
	done
	echo "digit sum: $sum"
}
reverse(){
	echo "Reverse order:"
	i=0
	while [ $i -lt 4 ];
	do
	n=$(($num2%10))
	echo $n

	((i++))
	done
	
}
