5.code:- --------------(input an expression and output the result)
	#include<stdio.h>
	int main(){
	int a,b;
	char op;

	printf("Enter expression (Eg.5+3): ");
	scanf("%d %c %d", &a, &op, &b);

	if (op== '+')
	printf("Result= %d\n",a+b);
	else if (op== '-')
	printf("Result = % \n", a-b);
	else if(op== '*')
	printf("Result= %d\n", a*b);
	else if(op == '/')
	printf("%d \n", a/b);
	else
	printf("Invalid input");
	return 0;
	}
	
output:-
	Enter expression (Eg.5+3): 6*4
	Result= 24
