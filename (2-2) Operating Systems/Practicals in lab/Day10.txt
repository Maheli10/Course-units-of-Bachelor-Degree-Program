03.code:(parent process:get a user input to find out factorial, 
		and in its child process: find factorial)
	#include<stdio.h>
	#include<unistd.h>
	int main(){
	int a;
	int fact=1;
	printf("Enter number: ");
	scanf("%d" ,&a);
	printf("Process id %d\n",getpid());
	if(fork()==0){
	for(int i=1; i<=a; i++){
	fact= fact*i;
	}
	printf("Factorial of %d is %d\n",a,fact);
	printf("(PID: %d,PPID: %d)\n",getpid(),getppid());
	}
	return 0;
	}
