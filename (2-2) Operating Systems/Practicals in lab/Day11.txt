#include<stdio.h>
#include<unistd.h>
#include<sys/msg.h>
#include<string.h>
#include<stdlib.h>

#define MSGKEY 1111

struct person {
    long mtype;   // message type (required)
    int age;
    char name[100];
    char regno[100];
};

int main() {

    int msgid;
    struct person p1, p2;

    int f = fork();

    if(f == 0) {

        // Child Process (Receiver)

        msgid = msgget(MSGKEY, IPC_CREAT | 0666);

        msgrcv(msgid, &p2, sizeof(p2) - sizeof(long), 1, 0);

        printf("\n--- Data Received ---\n");
        printf("Name   : %s\n", p2.name);
        printf("Reg No : %s\n", p2.regno);
        printf("Age    : %d\n", p2.age);

        // Delete message queue
        msgctl(msgid, IPC_RMID, NULL);
    }
    else {

        // Parent Process (Sender)

        msgid = msgget(MSGKEY, IPC_CREAT | 0666);

        p1.mtype = 1;

        printf("Enter name: ");
        scanf("%s", p1.name);

        printf("Enter regno: ");
        scanf("%s", p1.regno);

        printf("Enter age: ");
        scanf("%d", &p1.age);

        if(msgsnd(msgid, &p1, sizeof(p1) - sizeof(long), 0) < 0) {
            printf("Message not sent\n");
        }
        else {
            printf("\nMessage sent successfully\n");
        }
    }

    return 0;
}
