find all armstrong numbers from 100-999
An armstrong number of 3 digits is an integer such that sum of the cubes of its digits is equal to the number itself.
ex:
	3^3 + 7^3 + 1^3 = 371
	
	#include<stdio.h>
	int main(){
		for(int i= 100; i<1000; i++){
			int sum=0;
			while(i>0){
				int j= i%10;
				sum= sum+j;
				i=i/10;
			}
			if(sum== i){
			printf("%d is an armstrong",i);
			}
		}
		return 0;
	}
