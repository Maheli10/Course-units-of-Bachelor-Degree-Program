echo "Enter the four digit number: "
read num

digitSum() {
    temp1=$num
    sum=0

    while [ $temp1 -gt 0 ]
    do
        digit=$((temp1 % 10))
        sum=$((sum + digit))
        temp1=$((temp1 / 10))
    done

    echo "$sum"
}

reverse() {
    temp2=$num

    while [ $temp2 -gt 0 ]
    do
        digit=$((temp2 % 10))
        echo "$digit"
        temp2=$((temp2 / 10))
    done
}

echo "Digit sum:"
digitSum

echo "Reverse order:"
reverse
